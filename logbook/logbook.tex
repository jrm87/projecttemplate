% \smartinclude[type][path][caption][width][placement][label]
%   type      : raw | figure | table   (default = raw)
%   path      : file path to include (required)
%   caption   : optional caption (ignored for raw)
%   width     : width for graphics or resizebox (default = \linewidth)
%   placement : float placement for figure/table (default = H)
%   label     : optional label (e.g., fig:..., tab:...)
%
% Behavior:
%   - If file exists:
%       * raw    -> \input{path}
%       * figure -> figure env with \includegraphics[width=<width>]{path}, optional \caption
%       * table  -> table env with \resizebox{<width>}{!}{\input{path}}, optional \caption
%   - If file does not exist: prints a red inline warning.
%   - Does not need to escape the underscore es detokenize takes all the path literally. 
%
\NewDocumentCommand{\smartinclude}{ O{raw} m O{} O{\linewidth} O{H} O{} }{%
  \IfFileExists{\detokenize{#2}}{%
    % ------- FILE EXISTS -------
    \ifthenelse{\equal{#1}{raw}}{%
      \input{\detokenize{#2}}%
      \ifthenelse{\isempty{#6}}{}{\par\noindent\hbox{}\label{#6}}%
    }{%
      \ifthenelse{\equal{#1}{figure}}{%
        \begin{figure}[#5]
          \centering
          \includegraphics[width=#4]{\detokenize{#2}}%
          \ifthenelse{\isempty{#3}}{}{\caption{#3}}%
          \ifthenelse{\isempty{#6}}{}{\label{#6}}%
        \end{figure}
      }{%
        \ifthenelse{\equal{#1}{table}}{%
          \begin{table}[#5]
            \centering
            \resizebox{#4}{!}{\input{\detokenize{#2}}}%
            \ifthenelse{\isempty{#3}}{}{\caption{#3}}%
            \ifthenelse{\isempty{#6}}{}{\label{#6}}%
          \end{table}
        }{%
          % Fallback: unknown type → raw
          \input{\detokenize{#2}}%
          \ifthenelse{\isempty{#6}}{}{\par\noindent\hbox{}\label{#6}}%
        }%
      }%
    }%
  }{%
    % ------- FILE MISSING: safe message (underscores won't crash) -------
    \noindent\textcolor{red}{File \texttt{\detokenize{#2}} does not exist. Most likely, the task has not been run.}%
  }%
}
